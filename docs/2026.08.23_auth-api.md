### 2. Data Layer

#### AuthApi.kt
```kotlin
interface AuthApi {
    @POST("api/v1/auth/signup")
    suspend fun signup(@Body request: SignupRequest)

    @POST("api/v1/auth/verify-email")
    suspend fun verifyEmail(@Body request: VerifyEmailRequest)

    @POST("api/v1/auth/login")
    suspend fun login(@Body request: LoginRequest): LoginResponse

    @POST("api/v1/auth/logout")
    suspend fun logout()
}
```

#### DTOs
```kotlin
@Serializable data class SignupRequest(val email: String, val name: String, val password: String)
@Serializable data class VerifyEmailRequest(val email: String, val code: String)
@Serializable data class LoginRequest(val email: String, val password: String)
@Serializable data class LoginResponse(val accessToken: String, val refreshToken: String, val userId: String)
```

#### AuthRepositoryImpl.kt
```kotlin
@Singleton
class AuthRepositoryImpl @Inject constructor(
    private val authApi: AuthApi,
    private val tokenStorage: TokenStorage,
) : AuthRepository {

    override suspend fun signup(...): Result<Unit> =
        runCatching { authApi.signup(SignupRequest(email, name, password)) }

    override suspend fun verifyEmail(...): Result<Unit> =
        runCatching { authApi.verifyEmail(VerifyEmailRequest(email, code)) }

    override suspend fun login(...): Result<Boolean> = runCatching {
        val resp = authApi.login(LoginRequest(email, password))
        tokenStorage.accessToken  = resp.accessToken
        tokenStorage.refreshToken = resp.refreshToken
        tokenStorage.userId       = resp.userId
        tokenStorage.deviceId == null  // isFirst
    }

    override suspend fun logout(): Result<Unit> = runCatching {
        runCatching { authApi.logout() }  // 실패해도 로컬 토큰 삭제
        tokenStorage.clearAll()
    }
}
```

#### AuthModule.kt
```kotlin
@Module @InstallIn(SingletonComponent::class)
abstract class AuthModule {
    @Binds @Singleton
    abstract fun bindAuthRepository(impl: AuthRepositoryImpl): AuthRepository

    companion object {
        @Provides @Singleton
        fun provideAuthApi(@AppClient retrofit: Retrofit): AuthApi =
            retrofit.create(AuthApi::class.java)
    }
}
```

### 3. Presentation Layer

#### ViewModel UiState / Event 공통 패턴
각 ViewModel은 다음 패턴을 따른다:
- `data class UiState(val isLoading: Boolean, val errorMessage: String)`
- `_uiState: MutableStateFlow<UiState>` → `uiState: StateFlow<UiState>`
- `_events: Channel<Event>` → `events: Flow<Event>` (one-shot 네비게이션)

#### SplashViewModel.kt
```kotlin
@HiltViewModel
class SplashViewModel @Inject constructor(
    private val tokenStorage: TokenStorage,
) : ViewModel() {
    val hasSession: Boolean get() = tokenStorage.accessToken != null
}
```

#### SignupViewModel.kt
```kotlin
sealed class SignupEvent {
    data class NavigateToVerify(val email: String) : SignupEvent()
}

@HiltViewModel
class SignupViewModel @Inject constructor(private val signupUseCase: SignupUseCase) : ViewModel() {
    data class UiState(val isLoading: Boolean = false, val errorMessage: String = "")

    private val _uiState = MutableStateFlow(UiState())
    val uiState: StateFlow<UiState> = _uiState.asStateFlow()

    private val _events = Channel<SignupEvent>()
    val events: Flow<SignupEvent> = _events.receiveAsFlow()

    fun signup(email: String, name: String, password: String) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true, errorMessage = "") }
            signupUseCase(email, name, password)
                .onSuccess { _events.send(SignupEvent.NavigateToVerify(email)) }
                .onFailure { _uiState.update { s -> s.copy(isLoading = false, errorMessage = it.message ?: "회원가입에 실패했습니다.") } }
        }
    }

    fun clearError() = _uiState.update { it.copy(errorMessage = "") }
}
```

#### LoginViewModel.kt
```kotlin
sealed class LoginEvent {
    data class NavigateAfterLogin(val isFirst: Boolean) : LoginEvent()
}

@HiltViewModel
class LoginViewModel @Inject constructor(private val loginUseCase: LoginUseCase) : ViewModel() {
    data class UiState(val isLoading: Boolean = false, val errorMessage: String = "")

    private val _uiState = MutableStateFlow(UiState())
    val uiState: StateFlow<UiState> = _uiState.asStateFlow()

    private val _events = Channel<LoginEvent>()
    val events: Flow<LoginEvent> = _events.receiveAsFlow()

    fun login(email: String, password: String) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true, errorMessage = "") }
            loginUseCase(email, password)
                .onSuccess { isFirst -> _events.send(LoginEvent.NavigateAfterLogin(isFirst)) }
                .onFailure { _uiState.update { s -> s.copy(isLoading = false, errorMessage = it.message ?: "로그인에 실패했습니다.") } }
        }
    }

    fun clearError() = _uiState.update { it.copy(errorMessage = "") }
}
```

#### NavGraph.kt 변경 요점
```kotlin
composable(Screen.Splash.route) {
    val viewModel: SplashViewModel = hiltViewModel()
    SplashScreen(
        hasSession = viewModel.hasSession,
        onNavigateToLogin = { ... },
        onNavigateToList  = { ... },
    )
}

composable(Screen.Signup.route) {
    val viewModel: SignupViewModel = hiltViewModel()
    LaunchedEffect(viewModel) {
        viewModel.events.collect { event ->
            when (event) {
                is SignupEvent.NavigateToVerify ->
                    navController.navigate(Screen.Verify.createRoute(event.email))
            }
        }
    }
    SignupScreen(onBack = { navController.popBackStack() }, viewModel = viewModel)
}

composable(
    route = Screen.Verify.route,
    arguments = listOf(navArgument("email") { type = NavType.StringType; defaultValue = "" }),
) { backStackEntry ->
    val viewModel: VerifyViewModel = hiltViewModel()
    val email = backStackEntry.arguments?.getString("email").orEmpty()
    LaunchedEffect(viewModel) {
        viewModel.events.collect { event ->
            when (event) {
                VerifyEvent.NavigateToLogin ->
                    navController.navigate(Screen.Login.route) {
                        popUpTo(Screen.Signup.route) { inclusive = true }
                    }
            }
        }
    }
    VerifyScreen(email = email, onBack = { navController.popBackStack() }, viewModel = viewModel)
}

composable(Screen.Login.route) {
    val viewModel: LoginViewModel = hiltViewModel()
    LaunchedEffect(viewModel) {
        viewModel.events.collect { event ->
            when (event) {
                is LoginEvent.NavigateAfterLogin -> {
                    val dest = if (event.isFirst) Screen.Device.route else Screen.DestinationList.route
                    navController.navigate(dest) { popUpTo(Screen.Login.route) { inclusive = true } }
                }
            }
        }
    }
    LoginScreen(
        onNavigateToSignup = { navController.navigate(Screen.Signup.route) },
        viewModel = viewModel,
    )
}
```

## 에러 처리 전략

| 상황 | 처리 |
|------|------|
| API 2xx | onSuccess → 이벤트 발행 |
| API 4xx / 5xx | onFailure → errorMessage에 서버 메시지 또는 기본 문구 표시 |
| 네트워크 오류 | onFailure → 기본 문구 표시 |
| 토큰 갱신 실패 | TokenAuthenticator가 clearAll() 후 null 반환 → Phase 3에서는 미처리, Phase 3 이후 개선 |
