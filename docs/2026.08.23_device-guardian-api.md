# Phase 4+5 — Device·Guardian API 연동

## 개요

초기 설정 플로우의 두 번째·세 번째 화면인 DeviceScreen·GuardianScreen을 실제 API에 연결한다.  
Phase 4(기기 등록)와 Phase 5(보호자 등록)는 구조가 동일하고 NavGraph를 공유하므로 단일 브랜치 `feat/device-guardian-api`로 작업한다.

## Data Layer

### API

```kotlin
// DeviceApi.kt
interface DeviceApi {
    @POST("api/v1/devices")
    suspend fun registerDevice(@Body request: RegisterDeviceRequest): DeviceResponse

    @GET("api/v1/devices")
    suspend fun getDevices(): List<DeviceResponse>
}

// GuardianApi.kt
interface GuardianApi {
    @POST("api/v1/guardians")
    suspend fun registerGuardian(@Body request: RegisterGuardianRequest): GuardianResponse

    @GET("api/v1/guardians")
    suspend fun getGuardians(): List<GuardianResponse>
}
```

### DTO

```kotlin
@Serializable data class RegisterDeviceRequest(val deviceId: String)
@Serializable data class DeviceResponse(val deviceId: String, val name: String)

@Serializable data class RegisterGuardianRequest(val name: String, val phone: String)
@Serializable data class GuardianResponse(val guardianId: String, val name: String, val phone: String)
```

### RepositoryImpl

```kotlin
// DeviceRepositoryImpl.kt
@Singleton
class DeviceRepositoryImpl @Inject constructor(
    private val deviceApi: DeviceApi,
    private val tokenStorage: TokenStorage,
) : DeviceRepository {

    override suspend fun registerDevice(deviceId: String): Result<Device> = runCatching {
        val response = deviceApi.registerDevice(RegisterDeviceRequest(deviceId))
        tokenStorage.deviceId = response.deviceId   // TokenStorage에 저장
        Device(response.deviceId, response.name)
    }

    override suspend fun getDevices(): Result<List<Device>> = runCatching {
        deviceApi.getDevices().map { Device(it.deviceId, it.name) }
    }
}

// GuardianRepositoryImpl.kt
@Singleton
class GuardianRepositoryImpl @Inject constructor(
    private val guardianApi: GuardianApi,
) : GuardianRepository {

    override suspend fun registerGuardian(name: String, phone: String): Result<Guardian> =
        runCatching {
            val response = guardianApi.registerGuardian(RegisterGuardianRequest(name, phone))
            Guardian(response.guardianId, response.name, response.phone)
        }

    override suspend fun getGuardians(): Result<List<Guardian>> = runCatching {
        guardianApi.getGuardians().map { Guardian(it.guardianId, it.name, it.phone) }
    }
}
```

### Hilt 모듈

## Presentation Layer

### DeviceViewModel

```kotlin
sealed class DeviceEvent {
    object NavigateToGuardian : DeviceEvent()
}

@HiltViewModel
class DeviceViewModel @Inject constructor(
    private val registerDeviceUseCase: RegisterDeviceUseCase,
) : ViewModel() {
    data class UiState(val isLoading: Boolean = false, val errorMessage: String = "")

    private val _uiState = MutableStateFlow(UiState())
    val uiState: StateFlow<UiState> = _uiState.asStateFlow()

    private val _events = Channel<DeviceEvent>()
    val events: Flow<DeviceEvent> = _events.receiveAsFlow()

    fun register(deviceId: String) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true, errorMessage = "") }
            registerDeviceUseCase(deviceId)
                .onSuccess { _events.send(DeviceEvent.NavigateToGuardian) }
                .onFailure { error ->
                    _uiState.update { it.copy(isLoading = false, errorMessage = error.message ?: "기기 등록에 실패했습니다.") }
                }
        }
    }

    fun clearError() = _uiState.update { it.copy(errorMessage = "") }
}
```

### NavGraph 변경 요점

```kotlin
composable(Screen.Device.route) {
    val viewModel: DeviceViewModel = hiltViewModel()
    LaunchedEffect(viewModel) {
        viewModel.events.collect { event ->
            when (event) {
                DeviceEvent.NavigateToGuardian ->
                    navController.navigate(Screen.Guardian.route)
            }
        }
    }
    DeviceScreen(viewModel = viewModel)
}

composable(Screen.Guardian.route) {
    val viewModel: GuardianViewModel = hiltViewModel()
    LaunchedEffect(viewModel) {
        viewModel.events.collect { event ->
            when (event) {
                GuardianEvent.NavigateToDestinationList ->
                    navController.navigate(Screen.DestinationList.route) {
                        popUpTo(Screen.Device.route) { inclusive = true }
                    }
            }
        }
    }
    GuardianScreen(
        onSkip = {
            navController.navigate(Screen.DestinationList.route) {
                popUpTo(Screen.Device.route) { inclusive = true }
            }
        },
        viewModel = viewModel,
    )
}
```

## 에러 처리

| 상황 | 처리 |
|------|------|
| 등록 성공 | ViewModel 이벤트 → NavGraph 네비게이션 |
| 등록 실패 (4xx/5xx) | errorMessage 표시 |
| 보호자 건너뛰기 | onSkip() 직접 호출, tokenStorage 변경 없음 |
